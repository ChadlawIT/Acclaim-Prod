Select "REPORTS"."AGEDDEBT"."CLIENTNAME" As "Client",
  "REPORTS"."AGEDDEBT"."MTCODE" As "Our Reference",
  "PUB"."MATDB"."DESCRIPTION" As "Case Details",
  "REPORTS"."AGEDDEBT"."BILLNUMBER" As "Bill Number",
  "REPORTS"."AGEDDEBT"."BILLDATE" As "Bill Date",
  "REPORTS"."AGEDDEBT"."DAYSOUTST" As "Days Oustanding",
  Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" + "REPORTS"."AGEDDEBT"."AGEUPTO60" +
  "REPORTS"."AGEDDEBT"."AGEUPTO90" + "REPORTS"."AGEDDEBT"."AGEUPTO120" +
  "REPORTS"."AGEDDEBT"."AGEOVER120") As "Total Outstanding",
  "PUB"."MATAC"."BAL-CLI" As "[Client Balance]",
  Case
    When "PUB"."BILL-HEADER"."DISBS-NET" Is Not Null And
    "PUB"."BILL-HEADER"."DISBS-NET" <> 0 And
    Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" + "REPORTS"."AGEDDEBT"."AGEUPTO60" +
    "REPORTS"."AGEDDEBT"."AGEUPTO90" + "REPORTS"."AGEDDEBT"."AGEUPTO120" +
    "REPORTS"."AGEDDEBT"."AGEOVER120") = "PUB"."BILL-HEADER"."DISBS-NET" And
    "PUB"."MATAC"."BAL-CLI" = Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" +
    "REPORTS"."AGEDDEBT"."AGEUPTO60" + "REPORTS"."AGEDDEBT"."AGEUPTO90" +
    "REPORTS"."AGEDDEBT"."AGEUPTO120" + "REPORTS"."AGEDDEBT"."AGEOVER120") Then
    'Paid'
    When "PUB"."BILL-HEADER"."DISBS-NET" Is Not Null And
    "PUB"."BILL-HEADER"."DISBS-NET" <> 0 And
    Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" + "REPORTS"."AGEDDEBT"."AGEUPTO60" +
    "REPORTS"."AGEDDEBT"."AGEUPTO90" + "REPORTS"."AGEDDEBT"."AGEUPTO120" +
    "REPORTS"."AGEDDEBT"."AGEOVER120") = "PUB"."BILL-HEADER"."DISBS-NET" And
    "PUB"."MATAC"."BAL-CLI" > Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" +
    "REPORTS"."AGEDDEBT"."AGEUPTO60" + "REPORTS"."AGEDDEBT"."AGEUPTO90" +
    "REPORTS"."AGEDDEBT"."AGEUPTO120" + "REPORTS"."AGEDDEBT"."AGEOVER120") And
    "NT"."BILL-NO" = 'Pt Paid' Then 'Paid'
    When "PUB"."BILL-HEADER"."DISBS-NET" Is Not Null And
    "PUB"."BILL-HEADER"."DISBS-NET" <> 0 And
    Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" + "REPORTS"."AGEDDEBT"."AGEUPTO60" +
    "REPORTS"."AGEDDEBT"."AGEUPTO90" + "REPORTS"."AGEDDEBT"."AGEUPTO120" +
    "REPORTS"."AGEDDEBT"."AGEOVER120") = "PUB"."BILL-HEADER"."DISBS-NET" And
    "PUB"."MATAC"."BAL-CLI" > Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" +
    "REPORTS"."AGEDDEBT"."AGEUPTO60" + "REPORTS"."AGEDDEBT"."AGEUPTO90" +
    "REPORTS"."AGEDDEBT"."AGEUPTO120" + "REPORTS"."AGEDDEBT"."AGEOVER120") Then
    'Check' Else "CUSTOM"."CL0038"."AUTH2" End As "[AgedPaid]",
  "NT"."BILL-NO" As "[Nominal]",
  "CUSTOM"."CL0038"."BILLID" As "[Bill ID]",
  "PUB"."BILL-HEADER"."DISBS-NET" As "[CHECK Disbs on Bil]",
  "PUB"."BILL-HEADER"."COSTS-NET" As "[Cost on Bill]",
  "PUB"."BILL-HEADER"."COSTS-VAT" As "[VAT on Bill]",
  "REPORTS"."AGEDDEBT"."FEEEARNER" As "[Fee Earner Initials]"
From "REPORTS"."AGEDDEBT"
  Left Join "CUSTOM"."CL003A" On "REPORTS"."AGEDDEBT"."MTCODE" =
    "CUSTOM"."CL003A"."REFERENCE"
  Left Join "PUB"."MATDB" On "REPORTS"."AGEDDEBT"."MTCODE" =
    "PUB"."MATDB"."MT-CODE"
  Left Join "PUB"."MATAC" On "REPORTS"."AGEDDEBT"."MTCODE" =
    "PUB"."MATAC"."MT-CODE"
  Inner Join "PUB"."BILL-HEADER" On "PUB"."BILL-HEADER"."MT-CODE" =
    "REPORTS"."AGEDDEBT"."MTCODE" And "PUB"."BILL-HEADER"."BILL-NO" =
    "REPORTS"."AGEDDEBT"."BILLNUMBER"
  Inner Join "PUB"."MATTYPE" On "PUB"."MATDB"."MT-TYPE" =
    "PUB"."MATTYPE"."MT-TYPE"
  Inner Join "CUSTOM"."CL0038" On "PUB"."BILL-HEADER"."BILL-ID" =
    "CUSTOM"."CL0038"."BILLID"
  Left Join (Select "PUB"."NOMTRANS"."MT-CODE" As "MT-CODE",
    "PUB"."NOMTRANS"."TRANS-NO" As "TRANS-NO",
    Max("PUB"."NOMTRANS"."BILL-NO") As "BILL-NO"
  From "PUB"."NOMTRANS"
  Where "PUB"."NOMTRANS"."BILL-NO" Is Not Null
  Group By "PUB"."NOMTRANS"."MT-CODE",
    "PUB"."NOMTRANS"."TRANS-NO") "NT"
    On "NT"."MT-CODE" = "REPORTS"."AGEDDEBT"."MTCODE" And "NT"."TRANS-NO" =
    "PUB"."BILL-HEADER"."ACC-TRANS-NO"
Where "REPORTS"."AGEDDEBT"."DEPARTMENT" = 'LITI' And
  "PUB"."MATTYPE"."DEPARTMENT" = 'DEBT'
Group By "REPORTS"."AGEDDEBT"."CLIENTNAME",
  "REPORTS"."AGEDDEBT"."MTCODE",
  "PUB"."MATDB"."DESCRIPTION",
  "REPORTS"."AGEDDEBT"."BILLNUMBER",
  "REPORTS"."AGEDDEBT"."BILLDATE",
  "REPORTS"."AGEDDEBT"."DAYSOUTST",
  "PUB"."MATAC"."BAL-CLI",
  "NT"."BILL-NO",
  "CUSTOM"."CL0038"."BILLID",
  "PUB"."BILL-HEADER"."DISBS-NET",
  "PUB"."BILL-HEADER"."COSTS-NET",
  "PUB"."BILL-HEADER"."COSTS-VAT",
  "REPORTS"."AGEDDEBT"."FEEEARNER",
  "REPORTS"."AGEDDEBT"."DEPARTMENT",
  "CUSTOM"."CL003A"."ADSTATUS",
  "CUSTOM"."CL003A"."ADNOTES",
  "CUSTOM"."CL0038"."AUTH2",
  "PUB"."MATDB"."MT-TYPE",
  "PUB"."MATTYPE"."DEPARTMENT"
Having Sum("REPORTS"."AGEDDEBT"."AGEUPTO30" + "REPORTS"."AGEDDEBT"."AGEUPTO60" +
  "REPORTS"."AGEDDEBT"."AGEUPTO90" + "REPORTS"."AGEDDEBT"."AGEUPTO120" +
  "REPORTS"."AGEDDEBT"."AGEOVER120") > 0.00
Order By "Days Oustanding" Desc
